
class Solution {
    public int minimumWhiteTiles(String floor, int numCarpets, int carpetLen) {

        int n = floor.length();

        int[][] dp = new int[n + 1][numCarpets + 1];

        for (int i = n - 1; i >= 0; i--) {

            for (int k = 0; k <= numCarpets; k++) {

                // Choice 1: Don't cover current tile
                int skip = (floor.charAt(i) == '1' ? 1 : 0)
                           + dp[i + 1][k];

                // Choice 2: Cover current tile
                int cover = Integer.MAX_VALUE;

                if (k > 0) {
                    cover = dp[Math.min(n, i + carpetLen)][k - 1];
                }

                dp[i][k] = Math.min(skip, cover);
            }
        }

        return dp[0][numCarpets];
    }
}
