
class Solution {
    public int numOfSubarrays(int[] arr) {

        long even = 1;
        long odd = 0;
        long ans = 0;

        int parity = 0;

        for (int num : arr) {

            parity = (parity + num) % 2;

            if (parity == 1) {
                ans += even;
                odd++;
            } 
            else {
                ans += odd;
                even++;
            }

            ans %= 1000000007;
        }

        return (int) ans;
    }
}
